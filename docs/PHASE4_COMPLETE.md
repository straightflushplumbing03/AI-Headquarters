### Example Usage
```typescript
import { SupabaseClient } from '@supabase/supabase-js'
import { StraightFlushInitializer } from '@/lib/phase4/straight-flush-initializer'
import { ExecutiveDashboard } from '@/lib/phase4/executive-dashboard'
import { DailyBriefingSystem } from '@/lib/phase4/daily-briefing'

const supabase = new SupabaseClient(url, key)
const orgId = 'org-straightflush'

// Initialize organization
const init = new StraightFlushInitializer(supabase)
await init.initializeStraightFlush(orgId)

// Get daily briefing
const briefing = new DailyBriefingSystem(supabase)
const daily = await briefing.generateDailyBriefing(orgId)
console.log(daily.ceo_recommendation)

// Check executive state
const dashboard = new ExecutiveDashboard(supabase)
const state = await dashboard.getExecutiveState(orgId)
console.log(`Active workflows: ${state.active_workflows}`)
```
